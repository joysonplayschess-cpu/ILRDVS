_TRIM = " :-–—.|,/\\;,;_=~\"'`\t"

# Boilerplate values that crop up when a label spotter matches the *header*
# of the form (e.g. "Office of the Tehsildar", "…State Land Revenue Code").
_JUNK_VALUES = {"dar", "कार्यालय", "शासन", "विभाग", "सरकार", "government",
                "department", "office", "form", "prescribed", "code"}
_JUNK_HINT_RE = re.compile(
    r"(?i)form\s|prescribed|land\s*revenue|department|government|office\s+of")


# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------
def _clean_value(text: str) -> str:
    text = text.strip(_TRIM).strip()
    text = re.sub(r"\s{2,}", " ", text)
    text = _PREFIX_JUNK_RE.sub("", text)
    return text.strip(_TRIM).strip()


_CO_LABEL = re.compile(r"(?i)^(mauza|taluka|tehsil|tahsil|dist|zila|gram|gaon|"
                       r"village|state|district)\b\s*")

def _strip_colabels(value: str) -> str:
    """'/ Mauza Rampur' -> 'Rampur' style co-label residue cleanup."""
    value = _clean_value(value)
    for _ in range(2):
        m = _CO_LABEL.match(value)
        if not m:
            break
        value = _clean_value(value[m.end():])
    return value


def _smart_case(text: str) -> str:
    if text.isupper() or text.islower():
        return " ".join(w[:1].upper() + w[1:].lower() for w in text.split())
    return text


def _find_label_all(lines, field):
    """Return every (line, span, kind) candidate that mentions ``field``."""
    found = []
    for line in lines:
        for rx in _ALL_REGEX.get(field, []):
            m = rx.search(line["text"])
            if m:
                found.append((line, m.span(), "en"))
                break
    for line in lines:
        for token in INDIC_LABELS.get(field, []):
            idx = line["text"].find(token)
            if idx >= 0:
                cand = (line, (idx, idx + len(token)), "indic")
                if cand not in found:
                    found.append(cand)
                break
    return found


def _score_candidate(field, value, line, span):
    """Heuristic plausibility score for one label match."""
    s = 0.0
    if value:
        s += 2.0
        if len(value) >= 3:
            s += 1.0
    if value.strip().lower() in _JUNK_VALUES:
        s -= 5.0
    if field in ("state", "tehsil", "district", "village", "owner_name",
                 "father_name") and _JUNK_HINT_RE.search(value):
        s -= 5.0
    if span[1] <= max(30, int(len(line["text"]) * 0.45)):
        s += 1.0                       # labels in forms sit on the left
    return s


def _label_spans(line_text, exclude_field=None):
    """All label match spans in a line (used to truncate values)."""
    spans = []
    for field, rxs in _ALL_REGEX.items():
        for rx in rxs:
            for m in rx.finditer(line_text):
                spans.append((m.start(), m.end(), field))
    for field, tokens in INDIC_LABELS.items():
        for token in tokens:
            start = 0
            while True:
                idx = line_text.find(token, start)
                if idx < 0:
                    break
                spans.append((idx, idx + len(token), field))
                start = idx + 1
    spans.sort()
    if exclude_field:
        spans = [s for s in spans if s[2] != exclude_field]
    return spans


def _line_value(line, span, field):
    """Extract the value right of a matched label span within the line."""
    tail = line["text"][span[1]:]
    # cut the tail at the start of any *other* label
    cut = len(tail)
    for s, e, f in _label_spans(tail, exclude_field=field):
        if s <= cut:
            # guard: "pin" inside "pincode" style overlaps, require word break
            cut = min(cut, s)
            break
    value = _clean_value(tail[:cut])
    used_words = _words_after(line, span)
    return value, used_words


def _words_after(line, span):
    """Words of the line starting after character index ``span[1]``."""
    cur, words = 0, []
    for w in line["words"]:
        start = cur
        cur += len(w["text"]) + 1
        if start >= span[1]:
            words.append(w)
    return words


def _conf(words, label_bonus=0.05, base=None):
    if words:
        c = mean(min(w["conf"], 100) for w in words) / 100.0
    else:
        c = base if base is not None else 0.30
    return max(0.05, min(0.99, c + label_bonus))


def _parse_date(text):
    m = DATE_RE.search(text)
    if not m:
        return None
    d, mo, y = (int(m.group(i)) for i in (1, 2, 3))
    if y < 100:
        y += 2000 if y < 50 else 1900
    if not (1 <= mo <= 12 and 1 <= d <= 31 and 1850 <= y <= 2100):
        return None
    return f"{d:02d}/{mo:02d}/{y:04d}"


def _detect_unit(text):
    for name, hints in AREA_UNIT_HINTS:
        for hint in hints:
            if hint.isascii():
                if re.search("(?i)" + hint, text):
                    return name
            elif hint in text:
                return name
    return ""


def _parse_area(text, require_unit=False):
    """First plausible area number; when ``require_unit`` the number must be
    followed by an area unit (protects the free-text fallback from grabbing
    dates or register numbers)."""
    for m in NUM_RE.finditer(text):
        window = text[m.end():m.end() + 28]
        unit = _detect_unit(window)
        if require_unit and not unit:
            continue
        try:
            value = float(m.group(0).replace(",", ""))
        except ValueError:
            continue
        if not (0 < value < 1e7):
            continue
        return value, unit or _detect_unit(text)
    return None, ""


# ---------------------------------------------------------------------------
# Main extractor
# ---------------------------------------------------------------------------
def extract_fields(ocr: dict, apply_learning: bool = True) -> dict:
    """Turn OCR output into structured fields.

    Returns ``{field: {value, confidence, source, needs_review}}``.
    """
    from records.constants import REQUIRED_FIELDS

    lines = ocr["lines"]
    text = ocr["text"]
    threshold = settings.OCR_CONFIDENCE_THRESHOLD
    out = {}

    for field in FIELD_KEYS:
        candidates = _find_label_all(lines, field)
        value, source, conf, best_line, best_span = "", "ocr", 0.0, None, None
        best_score = None
        for line, span, kind in candidates:
            cand_value, words = _line_value(line, span, field)
            cand_conf = _conf(words, label_bonus=0.05)
            if not cand_value:                   # try the line below the label
                idx = lines.index(line)
                if idx + 1 < len(lines):
                    nxt = lines[idx + 1]
                    cand_value = _clean_value(nxt["text"])
                    cand_conf = _conf(nxt["words"], label_bonus=-0.05)
            score = _score_candidate(field, cand_value, line, span)
            if best_score is None or score > best_score:
                best_score, value, conf = score, cand_value, cand_conf
                best_line, best_span = line, span
        # ---- typed normalisation -------------------------------------
        if field == "plot_area":
            parsed, unit = (None, "")
            if value:
                parsed, unit = _parse_area(value)
            if parsed is None:
                parsed, unit = _parse_area(text, require_unit=True)
                if parsed is not None and not value:
                    conf = _conf(None, base=0.36)
            value = f"{parsed:g}" if parsed is not None else ""
            if unit:
                out["area_unit"] = {"value": unit, "confidence": conf,
                                    "source": "ocr",
                                    "needs_review": conf < threshold}
        elif field == "area_unit":
            value = out.get("area_unit", {}).get("value") or _detect_unit(text)
        elif field == "pincode":
            m = (PIN_RE.search(value) if value else None) or PIN_RE.search(text)
            value = m.group(0) if m else ""
            if value and not candidates:
                conf = _conf(None, base=0.42)
        elif field in ("survey_number", "khasra_number", "khata_number"):
            if value:
                m = PLOT_ID_RE.search(value.replace(" ", ""))
                value = m.group(0) if m else _clean_value(value)[:40]
        elif field in ("mutation_number", "registration_number"):
            if value:
                value = re.split(r"(?i)\bdate\b|दिनांक|दिनाक|दिनांक:" , value)[0]
                value = _clean_value(value)[:40]
        elif field in ("owner_name", "father_name", "land_classification",
                       "ownership_type"):
            value = _smart_case(_clean_value(value))[:120]
        elif field in ("village", "tehsil", "district", "state"):
            value = _smart_case(_strip_colabels(value))[:120]

        # ---- dates anchored to their labels --------------------------
        if field == "mutation_number" and best_line is not None:
            d = _parse_date(best_line["text"][best_span[1]:])
            if d:
                out["mutation_date"] = {"value": d, "confidence": conf,
                                        "source": "ocr",
                                        "needs_review": conf < threshold}
        if field == "registration_number" and best_line is not None:
            d = _parse_date(best_line["text"][best_span[1]:])
            if d:
                out["registration_date"] = {"value": d, "confidence": conf,
                                            "source": "ocr",
                                            "needs_review": conf < threshold}

        # ---- AI learning: auto-apply memorised corrections -----------
        if apply_learning and value and field in LEARNABLE_FIELDS:
            corrected = learn.apply_and_count(field, value)
            if corrected:
                value, source = corrected, "learned"
                conf = min(0.97, conf + 0.12)

        needs_review = conf < threshold or (field in REQUIRED_FIELDS and not value)

        if field in out and out[field]["value"]:
            continue                     # dates stored above keep priority
        out[field] = {"value": value, "confidence": round(conf, 3),
                      "source": source, "needs_review": bool(needs_review)}

    # backfill mutation/registration dates from the raw text when missing
    if not out.get("mutation_date", {}).get("value"):
        for line in lines:
            if any(t in line["text"] for t in ("दाखिल", "Mutation", "mutation")):
                d = _parse_date(line["text"])
                if d:
                    out["mutation_date"] = {"value": d,
                                            "confidence": _conf(line["words"], -0.05),
                                            "source": "ocr", "needs_review": False}
                    break
    if not out.get("registration_date", {}).get("value"):
        for line in lines:
            lt = line["text"].lower()
            if ("regist" in lt or "पंजीकरण"inlt or "रजिस्ट्री" in lt):
                d = _parse_date(line["text"])
                if d:
                    out["registration_date"] = {"value": d,
                                                "confidence": _conf(line["words"], -0.05),
                                                "source": "ocr", "needs_review": False}
                    break

    for f in FIELD_KEYS:        # guarantee full schema
        out.setdefault(f, {"value": "", "confidence": 0.0, "source": "ocr",
                           "needs_review": True})
    return out
